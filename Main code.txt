// ===== Hàm tiện ích cảm biến =====
static inline int readADCavg(int pin, int N=8) {
  long s = 0;
  for (int i=0; i<N; i++) { s += analogRead(pin); delayMicroseconds(200); }
  return (int)(s/N);
}

static inline float adcToVadc(int adc) { return (adc/4095.0f)*ADC_REF; }
static inline float vadcToVsensor(float vAdc) { return vAdc * AOUT_SCALE; }
static inline float calcRs_from_Vsensor(float vSen) {
  if (vSen < 0.005f) vSen = 0.005f;
  if (vSen > (VCC_SENSOR - 0.02f)) vSen = VCC_SENSOR - 0.02f;
  return RL_MODULE * (VCC_SENSOR / vSen - 1.0f); 
}

static inline float MQ_ppm_fromADC(int adc, float R0, float m, float b, float ppmMin, float ppmMax) {
  if (adc <= 0 || R0 < 1.0f) return 0.0f;
  float Vadc = adcToVadc(adc);
  float Vsen = vadcToVsensor(Vadc);
  float Rs   = calcRs_from_Vsensor(Vsen);
  float ratio = Rs / R0;
  if (ratio <= 0) return 0.0f;

  float ppm_log = m * log10f(ratio) + b;
  float ppm = powf(10.0f, ppm_log);
  if (!isfinite(ppm)) ppm = 0;
  if (ppm > ppmMax) ppm = ppmMax;
  return ppm;
}

// ===== Hàm lấy thời gian =====
String getDateTimeString() {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) {
    return "No Time";
  }
  char buf[30];
  strftime(buf, sizeof(buf), "%d/%m/%Y %H:%M", &timeinfo);
  return String(buf);
}

// ===== Hàm xử lý nút nhấn =====
bool checkButton(int pin) {
  static unsigned long lastDebounce[50] = {0};
  static int lastState[50] = {LOW};
  bool pressed = false;
  int reading = digitalRead(pin);
  if (reading != lastState[pin]) {
    lastDebounce[pin] = millis();
  }
  if ((millis() - lastDebounce[pin]) > 50) {
    if (lastState[pin] == LOW && reading == HIGH) {
      pressed = true;
    }
  }
  lastState[pin] = reading;
  return pressed;
}

// ===== Hàm đọc và xử lý dữ liệu cảm biến =====
void readSensors(float &t, float &h, float &mq2_Smoke, float &mq2_LPG, float &mq2_H2, 
                 float &mq5_CH4, float &mq5_LPG, float &mq9_CO, float &mq9_CH4) {
  t = dht.readTemperature();
  h = dht.readHumidity();

  int a2 = readADCavg(MQ2_PIN);
  int a5 = readADCavg(MQ5_PIN);
  int a9 = readADCavg(MQ9_PIN);

  mq2_Smoke = MQ_ppm_fromADC(a2, R0_MQ2, m_MQ2_Smoke, b_MQ2_Smoke, 10, 10000);
  mq2_LPG   = MQ_ppm_fromADC(a2, R0_MQ2, m_MQ2_LPG,   b_MQ2_LPG,   10, 10000);
  mq2_H2    = MQ_ppm_fromADC(a2, R0_MQ2, m_MQ2_H2,    b_MQ2_H2,    10, 10000);
  mq5_CH4   = MQ_ppm_fromADC(a5, R0_MQ5, m_MQ5_CH4,   b_MQ5_CH4,   10, 10000);
  mq5_LPG   = MQ_ppm_fromADC(a5, R0_MQ5, m_MQ5_LPG,   b_MQ5_LPG,   10, 10000);
  mq9_CH4   = MQ_ppm_fromADC(a9, R0_MQ9, m_MQ9_CH4,   b_MQ9_CH4,   10, 10000);
  mq9_CO    = MQ_ppm_fromADC(a9, R0_MQ9, m_MQ9_CO,    b_MQ9_CO,    10, 1000);

  Serial.println("===== SENSOR DATA =====");
  Serial.printf("DHT22: Temp=%.1f C  Hum=%.1f %%\n", t, h);
  Serial.printf("MQ2 Smoke=%.1f ppm, LPG=%.1f ppm, H2=%.1f ppm\n", mq2_Smoke, mq2_LPG, mq2_H2);
  Serial.printf("MQ5 CH4=%.1f ppm, LPG=%.1f ppm\n", mq5_CH4, mq5_LPG);
  Serial.printf("MQ9 CO=%.1f ppm, CH4=%.1f ppm\n", mq9_CO, mq9_CH4);
  Serial.println("========================");
}

// ===== Hàm tính toán mức nguy cơ bằng M ngưỡng =====
void calculateRisk(float t, float mq2_Smoke, float mq2_LPG, float mq5_CH4, float mq5_LPG, 
                   float mq9_CO, float mq9_CH4, float &M_fire, float &M_gas, float &M_khoi) {
                  
  //Ngưỡng cháy : Nhiệt độ DHT 35% + MQ2 khói 20% + MQ9 CO 30% + ( LPG+CH4) 5%
  M_fire = 0.35f  * (t / 80.0f * 100.0f)            
         + 0.30f  * (mq2_Smoke / 10000.0f * 100.0f)
         + 0.30f  * (mq9_CO / 1000.0f * 100.0f)
         + 0.025f * (mq2_LPG / 10000.0f * 100.0f)
         + 0.025f * (mq5_CH4 / 10000.0f * 100.0f);
  M_fire = constrain(M_fire, 0.0f, 100.0f);

  //Ngưỡng gas : MQ5 (CH4+LPG) 60% + MQ2 LPG 25% + MQ9 CH4 15%
  M_gas = 0.40f * (mq5_CH4 / 10000.0f * 100.0f)
        + 0.20f * (mq5_LPG / 10000.0f * 100.0f)
        + 0.25f * (mq2_LPG / 10000.0f * 100.0f)
        + 0.15f * (mq9_CH4 / 10000.0f * 100.0f);
  M_gas = constrain(M_gas, 0.0f, 100.0f);

  //Ngưỡng khói : MQ2 Smoke 100%
  M_khoi = (mq2_Smoke / 10000.0f) * 100.0f;
  M_khoi = constrain(M_khoi, 0.0f, 100.0f);
}

// ===== Hàm cập nhật giao diện TFT =====
void updateTFT(float t, float h, float mq2_Smoke, float mq2_LPG, float mq2_H2, 
               float mq5_CH4, float mq5_LPG, float mq9_CO, float mq9_CH4, 
               String warning) {
  static float lastTemp=-1000, lastHum=-1000;
  static float lastSmoke=-1, lastLPG2=-1, lastH2=-1;
  static float lastCH4_5=-1, lastLPG5=-1, lastCO9=-1, lastCH4_9=-1;
  static String lastWarning="", lastDatetime="", lastModeStr="";
  static int lastRelayState[4] = {-1,-1,-1,-1};

  int y = yBase;

  updateText(xVal, y, getDateTimeString(), lastDatetime, ST77XX_WHITE); y += lineHeight;
  updateText(xVal, y, mode ? "Manual" : "Auto", lastModeStr, ST77XX_YELLOW); y += lineHeight;
  updateValue(xVal, y, t, lastTemp, "%.1f C", ST77XX_RED); y += lineHeight;
  updateValue(xVal, y, h, lastHum, "%.1f %%", ST77XX_BLUE); y += lineHeight;
  updateValue(xVal, y, mq2_Smoke, lastSmoke, "%.0f ppm", ST77XX_CYAN); y += lineHeight;
  updateValue(xVal, y, mq2_LPG, lastLPG2, "%.0f ppm", ST77XX_CYAN); y += lineHeight;
  updateValue(xVal, y, mq2_H2, lastH2, "%.0f ppm", ST77XX_CYAN); y += lineHeight;
  updateValue(xVal, y, mq5_CH4, lastCH4_5, "%.0f ppm", ST77XX_CYAN); y += lineHeight;
  updateValue(xVal, y, mq5_LPG, lastLPG5, "%.0f ppm", ST77XX_CYAN); y += lineHeight;
  updateValue(xVal, y, mq9_CO, lastCO9, "%.0f ppm", ST77XX_CYAN); y += lineHeight;
  updateValue(xVal, y, mq9_CH4, lastCH4_9, "%.0f ppm", ST77XX_CYAN); y += lineHeight;

  for (int i=0; i<4; i++) {
    int relayPin = (i==0)?RELAY1:(i==1)?RELAY2:(i==2)?RELAY3:RELAY4;
    int state = digitalRead(relayPin);
    if (state != lastRelayState[i]) {
      tft.fillRect(xVal, y, 100, 15, ST77XX_BLACK);
      tft.setCursor(xVal, y);
      tft.setTextColor(state ? ST77XX_GREEN : ST77XX_RED, ST77XX_BLACK);
      tft.print(state ? "ON" : "OFF");
      lastRelayState[i] = state;
    }
    y += lineHeight;
  }

  uint16_t warnColor = ST77XX_GREEN;
  if (warning.indexOf("Khan Cap") >= 0 || warning.indexOf("Nguy Hiem") >= 0) warnColor = ST77XX_RED;
  else if (warning.indexOf("Theo Doi") >= 0) warnColor = ST77XX_YELLOW;
  updateText(xVal, y, warning, lastWarning, warnColor);
}

    if (newMode != mode) {

        digitalWrite(RELAY2, relay2State ? HIGH : LOW);

        digitalWrite(RELAY3, relay3State ? HIGH : LOW);

void syncThresholdsFromFirebase() {
  if (!Firebase.ready()) return;
  if (Firebase.RTDB.getJSON(&firebaseData, "users/0776707441/123456/ESP32/Threshold")) {
    FirebaseJson &json = firebaseData.jsonObject();
    size_t len = json.iteratorBegin();
    for (size_t i = 0; i < len; i++) {
      String key, value;
      int type;
      json.iteratorGet(i, type, key, value);

      if (key.startsWith("fire")) {
        int idx = key.substring(4).toInt();
        if (idx >= 1 && idx <= 4) thresh_fire[idx] = value.toInt();
      } else if (key.startsWith("khoi")) {
        int idx = key.substring(4).toInt();
        if (idx >= 1 && idx <= 4) thresh_khoi[idx] = value.toInt();
      } else if (key.startsWith("gas")) {
        int idx = key.substring(3).toInt();
        if (idx >= 1 && idx <= 4) thresh_gas[idx] = value.toInt();
      }
    }
    json.iteratorEnd();
    Serial.println(" Đã cập nhật Threshold từ Firebase");
  }
}

void sendToFirebase(float t, float h, float mq2_Smoke, float mq2_LPG, float mq2_H2, 
                    float mq5_CH4, float mq5_LPG, float mq9_CO, float mq9_CH4, 
                    float M_fire, float M_gas, float M_khoi, String warning, String title) {
  json.clear();
  json.set("temperature", t);
  json.set("humidity", h);
  json.set("mq2_Smoke", mq2_Smoke);
  json.set("mq2_LPG", mq2_LPG);
  json.set("mq2_H2", mq2_H2);
  json.set("mq5_LPG", mq5_LPG);
  json.set("mq5_CH4", mq5_CH4);
  json.set("mq9_CO", mq9_CO);
  json.set("mq9_CH4", mq9_CH4);
  json.set("M_fire", M_fire);
  json.set("M_gas", M_gas);
  json.set("M_khoi", M_khoi);
  json.set("warning", warning);
  json.set("title", title);
  json.set("Mode", mode ? "Manual" : "Auto");
  json.set("Relay1", digitalRead(RELAY1) ? "ON" : "OFF");
  json.set("Relay2", digitalRead(RELAY2) ? "ON" : "OFF");
  json.set("Relay3", digitalRead(RELAY3) ? "ON" : "OFF");
  json.set("Relay4", digitalRead(RELAY4) ? "ON" : "OFF");

  String path = "users/0776707441/123456/ESP32/Data";
  if (!Firebase.RTDB.setJSON(&firebaseData, path.c_str(), &json)) {
    Serial.println("Firebase error: " + firebaseData.errorReason());
  } else {
    Serial.println("Firebase update OK");
  }
}


void setup() {
  initHardware();
  initWiFi();
  initFirebase();
  Serial.println("Warm-up 10s...");
  delay(10000);
}

void loop() {
  unsigned long now = millis();

  // Xử lý nút nhấn
  handleButtons();

  // Kiểm tra WiFi
  if (WiFi.status() != WL_CONNECTED) {
    if (wifiConnected) {
      showWiFiStatus("Error WiFi", ST77XX_RED);
      wifiConnected = false;
      digitalWrite(RELAY1, LOW);
      digitalWrite(RELAY2, LOW);
      digitalWrite(RELAY3, LOW);
      digitalWrite(RELAY4, LOW);
    }
    return;
  } else if (!wifiConnected) {
    showWiFiStatus("Connection success", ST77XX_GREEN);
    delay(1500);
    drawUI();
    wifiConnected = true;
  }

  // Đọc cảm biến và cập nhật mỗi 3 giây
  if (now - lastSensorRead >= 3000) {
    lastSensorRead = now;

    float t, h, mq2_Smoke, mq2_LPG, mq2_H2, mq5_CH4, mq5_LPG, mq9_CO, mq9_CH4;
    readSensors(t, h, mq2_Smoke, mq2_LPG, mq2_H2, mq5_CH4, mq5_LPG, mq9_CO, mq9_CH4);

    float M_fire, M_gas, M_khoi;
    calculateRisk(t, mq2_Smoke, mq2_LPG, mq5_CH4, mq5_LPG, mq9_CO, mq9_CH4, M_fire, M_gas, M_khoi);

    String warning, title;
    getWarningAndTitle(M_fire, M_gas, M_khoi, warning, title);

    if (mode == 0) {
      controlRelays(warning);
    }

    updateTFT(t, h, mq2_Smoke, mq2_LPG, mq2_H2, mq5_CH4, mq5_LPG, mq9_CO, mq9_CH4, warning);
    syncFirebaseModeAndRelays();

    // Đọc và cập nhật ngưỡng mỗi 10 giây
    if (now - lastThresholdSync >= 10000) { 
          syncThresholdsFromFirebase();
          lastThresholdSync = now;
        }
    
    //Cập nhật dữ liệu lên Firebase mỗi 3 giây
    if (now - lastFirebaseSend >= 3000) {
      lastFirebaseSend = now;
      if (Firebase.ready()) {
        sendToFirebase(t, h, mq2_Smoke, mq2_LPG, mq2_H2, mq5_CH4, mq5_LPG, mq9_CO, mq9_CH4, 
                       M_fire, M_gas, M_khoi, warning, title);
      }
    }
  }
}
