/*================================= 7. HÀM TIỆN ÍCH CẢM BIẾN ===========================*/
float safeFloat(float v, float fallback = 0.0f) {
  if (isnan(v) || !isfinite(v)) return fallback;
  return v;
}

static inline int readADCavg(int pin, int N = 5) {
  long sum = 0;
  for (int i = 0; i < N; i++) {
    sum += analogRead(pin);
    delayMicroseconds(200);
  }
  return (int)(sum / N);
}

static inline float adcToVadc(int adc) {
  return (adc / 4095.0f) * ADC_REF;
}

static inline float vadcToVsensor(float vAdc) {
  return vAdc * AOUT_SCALE;
}

static inline float calcRs_from_Vsensor(float vSen) {
  if (vSen < 0.005f) vSen = 0.005f;
  if (vSen > (VCC_SENSOR - 0.02f)) vSen = VCC_SENSOR - 0.02f;
  return RL_MODULE * (VCC_SENSOR / vSen - 1.0f);
}

static inline float MQ_ppm_fromADC(int adc, float R0, float m, float b, float ppmMin, float ppmMax) {
  if (adc <= 0 || R0 < 1.0f) return 0.0f;
  float Vadc = adcToVadc(adc);
  float Vsen = vadcToVsensor(Vadc);
  float Rs   = calcRs_from_Vsensor(Vsen);
  float ratio = Rs / R0;
  if (ratio <= 0) return 0.0f;
  float ppm_log = m * log10f(ratio) + b;
  float ppm = powf(10.0f, ppm_log);
  if (!isfinite(ppm)) ppm = 0;
  ppm = constrain(ppm, ppmMin, ppmMax);
  return ppm;
}

String getDateString() {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) return "Date";
  char buf[20];
  strftime(buf, sizeof(buf), "%d/%m/%Y", &timeinfo);
  return String(buf);
}

String getTimeString() {
  struct tm timeinfo;
  if (!getLocalTime(&timeinfo)) return "Time";
  char buf[20];
  strftime(buf, sizeof(buf), "%H:%M:%S", &timeinfo);
  return String(buf);
}

/*============================= 11. XỬ LÝ NÚT NHẤN & RELAY =============================*/
void handleButtons() {
  static bool lastBtn1 = false, lastBtn2 = false, lastBtn3 = false, lastBtn4 = false, lastMode = false;
  bool stBtn1 = !digitalRead(BTN_RELAY1);
  bool stBtn2 = !digitalRead(BTN_RELAY2);
  bool stBtn3 = !digitalRead(BTN_RELAY3);
  bool stBtn4 = !digitalRead(BTN_RELAY4);
  bool stMode = !digitalRead(BTN_MODE);

  if (!digitalRead(BTN_RESET)) {
    Serial.println("Reset pressed!");
    tft.fillScreen(ST77XX_BLACK);
    tft.setTextColor(ST77XX_RED); tft.setTextSize(2);
    tft.setCursor(20, 120); tft.println("Restarting...");
    delay(1000);
    ESP.restart();
  }

  if (!lastMode && stMode) {
    mode = !mode;
    Serial.printf("Mode: %s\n", mode ? "Manual" : "Auto");
    if (Firebase.ready()) {
      Firebase.RTDB.setString(&firebaseData, "ESP32/Control/Mode", mode ? "Manual" : "Auto");
    }
    relay1State = relay2State = relay3State = relay4State = false;
    digitalWrite(RELAY1, LOW); digitalWrite(RELAY2, LOW);
    digitalWrite(RELAY3, LOW); digitalWrite(RELAY4, LOW);
    delay(200);
  }
  lastMode = stMode;

  if (mode == 1) {
    if (!lastBtn1 && stBtn1) { relay1State = !relay1State; digitalWrite(RELAY1, relay1State); delay(100); }
    if (!lastBtn2 && stBtn2) { relay2State = !relay2State; digitalWrite(RELAY2, relay2State); delay(100); }
    if (!lastBtn3 && stBtn3) { relay3State = !relay3State; digitalWrite(RELAY3, relay3State); delay(100); }
    if (!lastBtn4 && stBtn4) { relay4State = !relay4State; digitalWrite(RELAY4, relay4State); delay(100); }
  }

  lastBtn1 = stBtn1; lastBtn2 = stBtn2; lastBtn3 = stBtn3; lastBtn4 = stBtn4;
}

void controlRelays(float M_fire, float M_gas) {

  // ===== 1. Xác định mức nguy hiểm riêng cho từng loại =====
  bool fireDanger = (M_fire >= thresh_fire[2]);
  bool gasDanger  = (M_gas >= thresh_gas[2]);   

  // ===== 2. Xử lý từng trường hợp =====
  if (fireDanger && gasDanger) {
    digitalWrite(RELAY1, HIGH);  // Còi hú
    digitalWrite(RELAY2, HIGH);  // Quạt 1
    digitalWrite(RELAY3, HIGH);  // Quạt 2
    digitalWrite(RELAY4, HIGH);  // Máy bơm
  }

  else if (fireDanger) {
    digitalWrite(RELAY1, HIGH);  // Còi hú
    digitalWrite(RELAY2, LOW);
    digitalWrite(RELAY3, LOW);
    digitalWrite(RELAY4, HIGH);  // Máy bơm
  }

  else if (gasDanger) {
    digitalWrite(RELAY1, HIGH);  // Còi hú
    digitalWrite(RELAY2, HIGH);  // Quạt 1
    digitalWrite(RELAY3, HIGH);  // Quạt 2
    digitalWrite(RELAY4, LOW);
  }
  else {
    digitalWrite(RELAY1, LOW);
    digitalWrite(RELAY2, LOW);
    digitalWrite(RELAY3, LOW);
    digitalWrite(RELAY4, LOW);
  }
}

  static float camp_co    = 0.0f;
  static float camp_hum   = 0.0f;

  static float camp_lpg   = 0.0f;

  // ======== Giới hạn và chuẩn hóa từng giá trị (0–100%) ========
  float temp_part  = (min(t, 80.0f) / 80.0f) * 100.0f;
  float smoke_part = (min(mq2_Smoke, 1000.0f) / 1000.0f) * 100.0f;
  float co_part    = (min(mq7_CO, 400.0f) / 400.0f) * 100.0f;
  float hum_part   = ((100.0f - min(h, 100.0f)) / 100.0f) * 100.0f;
  float ch4_part   = (min(mq5_CH4, 3000.0f) / 3000.0f) * 100.0f;
  float lpg_part   = (min(mq5_LPG, 2000.0f) / 2000.0f) * 100.0f;

  // ======== Lọc mượt từng thành phần ========
  camp_temp  = camp_temp  * (1 - alpha) + temp_part  * alpha;
  camp_smoke = camp_smoke * (1 - alpha) + smoke_part * alpha;
  camp_co    = camp_co    * (1 - alpha) + co_part    * alpha;
  camp_hum   = camp_hum   * (1 - alpha) + hum_part   * alpha;
  camp_ch4   = camp_ch4   * (1 - alpha) + ch4_part   * alpha;
  camp_lpg   = camp_lpg   * (1 - alpha) + lpg_part   * alpha;

  // ======== 1. Mức nguy hiểm cháy ========
  M_fire = w_fire_temp  * camp_temp
         + w_fire_smoke * camp_smoke
         + w_fire_co    * camp_co
         + w_fire_hum   * camp_hum
         + w_fire_ch4   * camp_ch4
         + w_fire_lpg   * camp_lpg;

  M_fire = constrain(M_fire, 0.0f, 100.0f);

  // ======== CAMP riêng cho phần gas ========
  static float camp_ch4_gas = 0.0f;
  static float camp_lpg_gas = 0.0f;
  static float camp_co_gas  = 0.0f;

  camp_ch4_gas = camp_ch4_gas * (1 - alpha) + ch4_part * alpha;
  camp_lpg_gas = camp_lpg_gas * (1 - alpha) + lpg_part * alpha;
  camp_co_gas  = camp_co_gas  * (1 - alpha) + co_part  * alpha;

  // ======== 2. Mức nguy hiểm khí gas ========
  M_gas = w_gas_ch4 * camp_ch4_gas
        + w_gas_lpg * camp_lpg_gas
        + w_gas_co  * camp_co_gas;

  M_gas = constrain(M_gas, 0.0f, 100.0f);
}


void addAnalysisLevel(String &analysis, float value, float vlow, float low, float high, float vhigh,
                      const char* name, const char* unit) {
  String part = "";
  if (value >= vhigh) part = String(name) + " rất cao (" + String(value, (unit[0] == '%') ? 1 : 0) + unit + ")";
  else if (value >= high) part = String(name) + " cao (" + String(value, (unit[0] == '%') ? 1 : 0) + unit + ")";
  else if (value >= low) part = String(name) + " thấp (" + String(value, (unit[0] == '%') ? 1 : 0) + unit + ")";
  else if (value >= vlow) part = String(name) + " rất thấp (" + String(value, (unit[0] == '%') ? 1 : 0) + unit + ")";
  if (part != "" && analysis != "") analysis += ", ";
  if (part != "") analysis += part;
}

String getDetailedFireAnalysis(float t, float h, float smoke, float ch4, float lpg, float co, float M_fire) {
  String analysis = "";
  addAnalysisLevel(analysis, t,     threshold.temp_very_low,  threshold.temp_low,  threshold.temp_high,  threshold.temp_very_high,  "Nhiệt độ", "°C");
  addAnalysisLevel(analysis, h,     threshold.hum_very_low,   threshold.hum_low,   threshold.hum_high,   threshold.hum_very_high,   "Độ ẩm", "%");
  addAnalysisLevel(analysis, smoke, threshold.smoke_very_low, threshold.smoke_low, threshold.smoke_high, threshold.smoke_very_high, "Khói", " ppm");
  addAnalysisLevel(analysis, co,    threshold.co_very_low,    threshold.co_low,    threshold.co_high,    threshold.co_very_high,    "Khí CO", " ppm");
  addAnalysisLevel(analysis, ch4,   threshold.ch4_very_low,  threshold.ch4_low,   threshold.ch4_high,   threshold.ch4_very_high,   "Khí CH4", " ppm");
  addAnalysisLevel(analysis, lpg,   threshold.lpg_very_low,  threshold.lpg_low,   threshold.lpg_high,   threshold.lpg_very_high,   "Khí LPG", " ppm");

  if (analysis != "") analysis += ", ";
  if (M_fire >= thresh_fire[4]) analysis += "Mức cháy: Khẩn cấp";
  else if (M_fire >= thresh_fire[3]) analysis += "Mức cháy: Rất cao";
  else if (M_fire >= thresh_fire[2]) analysis += "Mức cháy: Cao";
  else if (M_fire >= thresh_fire[1]) analysis += "Mức cháy: Cần theo dõi";
  else analysis += "Mức cháy: An toàn";

  if (analysis == "") analysis = "Không phát hiện bất thường.";
  analysis.trim();
  return analysis;
}

/*================================= 14. ĐỒNG BỘ FIREBASE ===============================*/
void syncFirebaseModeAndRelays() {
  if (!Firebase.ready()) return;

  // Đọc node ESP32/Control 1 lần duy nhất
  if (!Firebase.RTDB.getJSON(&firebaseData, "ESP32/Control")) return;
  FirebaseJson &json = firebaseData.jsonObject();
  FirebaseJsonData data;

  // ==== MODE ====
  if (json.get(data, "Mode") && data.type == "string") {
    String fbMode = data.stringValue;
    int newMode = (fbMode == "Manual") ? 1 : 0;
    if (newMode != mode) {
      mode = newMode;
      // Khi đổi mode -> reset toàn bộ relay để tránh trạng thái sai
      relay1State = relay2State = relay3State = relay4State = false;
      digitalWrite(RELAY1, LOW);
      digitalWrite(RELAY2, LOW);
      digitalWrite(RELAY3, LOW);
      digitalWrite(RELAY4, LOW);
    }
  }

  // ==== RELAYS ====
  if (mode == 1) { // chỉ đồng bộ khi Manual
    String keys[4] = {"Relay1", "Relay2", "Relay3", "Relay4"};
    bool* relayStates[4] = {&relay1State, &relay2State, &relay3State, &relay4State};
    int relayPins[4] = {RELAY1, RELAY2, RELAY3, RELAY4};
    String* fbRelays[4] = {&fbRelay1, &fbRelay2, &fbRelay3, &fbRelay4};

    for (int i = 0; i < 4; i++) {
      if (json.get(data, keys[i].c_str()) && data.type == "string") {
        String v = data.stringValue;
        // Kiểm tra giá trị hợp lệ: chỉ nhận "ON"/"OFF"
        if (v == "ON" || v == "OFF") {
          if (v != *fbRelays[i]) {
            *relayStates[i] = (v == "ON");
            digitalWrite(relayPins[i], *relayStates[i]);
            *fbRelays[i] = v;
          }
        }
      }
    }
  }
}

void sendToFirebase(float t, float h, float mq2_Smoke, float mq5_CH4, float mq5_LPG, float mq7_CO,
                    float M_fire, float M_gas) {
  String currentDate = getDateString();
  String currentTime = getTimeString();
  t         = safeFloat(t);
  h         = safeFloat(h);
  mq2_Smoke = safeFloat(mq2_Smoke);
  mq5_CH4   = safeFloat(mq5_CH4);
  mq5_LPG   = safeFloat(mq5_LPG);
  mq7_CO    = safeFloat(mq7_CO);
  M_fire    = safeFloat(M_fire);
  M_gas     = safeFloat(M_gas);

  json.clear();
  json.set("date", currentDate);
  json.set("time", currentTime);
  json.set("temperature", t);
  json.set("humidity", h);
  json.set("mq2_Smoke", mq2_Smoke);
  json.set("mq5_CH4", mq5_CH4);
  json.set("mq5_LPG", mq5_LPG);
  json.set("mq7_CO", mq7_CO);
  json.set("M_fire", M_fire);
  json.set("M_gas", M_gas);
  json.set("Warning_fire", getFireWarning(M_fire));
  json.set("Warning_gas", getGasWarning(M_gas));
  json.set("Detailed_Analysis_Fire", getDetailedFireAnalysis(t, h, mq2_Smoke, mq5_CH4, mq5_LPG, mq7_CO, M_fire));
  json.set("Detailed_Analysis_Gas", getDetailedGasAnalysis(mq5_CH4, mq5_LPG, mq7_CO, M_gas));
  json.set("Mode", mode ? "Manual" : "Auto");
  json.set("Relay1", digitalRead(RELAY1) ? "ON" : "OFF");
  json.set("Relay2", digitalRead(RELAY2) ? "ON" : "OFF");
  json.set("Relay3", digitalRead(RELAY3) ? "ON" : "OFF");
  json.set("Relay4", digitalRead(RELAY4) ? "ON" : "OFF");

  if (Firebase.RTDB.updateNode(&firebaseData, "ESP32/Data", &json)) {
    Serial.printf("Firebase updated: %s %s\n", currentDate.c_str(), currentTime.c_str());
  } else {
    Serial.println("Firebase error: " + firebaseData.errorReason());
  }
}

void readSensors(float &t, float &h, float &mq2_Smoke, float &mq5_CH4, float &mq5_LPG, float &mq7_CO) {
  // Đọc cảm biến DHT
  t = dht.readTemperature();
  h = dht.readHumidity();

  // Nếu lỗi thì giữ giá trị cũ hoặc đặt fallback = 0
  static float lastT = 30.0f, lastH = 70.0f;
  if (isnan(t)) t = lastT;
  else lastT = t;

  if (isnan(h)) h = lastH;
  else lastH = h;

  // Đọc ADC MQ
  int a2 = readADCavg(MQ2_PIN);
  int a5 = readADCavg(MQ5_PIN);
  int a7 = readADCavg(MQ7_PIN);

  mq2_Smoke = safeFloat(MQ_ppm_fromADC(a2, R0_MQ2, m_MQ2_Smoke, b_MQ2_Smoke, 10, 1000));
  mq5_CH4   = safeFloat(MQ_ppm_fromADC(a5, R0_MQ5, m_MQ5_CH4,   b_MQ5_CH4,   10, 3000));
  mq5_LPG   = safeFloat(MQ_ppm_fromADC(a5, R0_MQ5, m_MQ5_LPG,   b_MQ5_LPG,   10, 2000));
  mq7_CO    = safeFloat(MQ_ppm_fromADC(a7, R0_MQ7, m_MQ7_CO,    b_MQ7_CO,    1, 400));
}

void loop() {
  unsigned long now = millis();
  handleButtons();

  // === 1. Kiểm tra trạng thái WiFi ===
  bool wifiOK = (WiFi.status() == WL_CONNECTED);

  // Nếu mất WiFi → hiển thị thông báo và nhấp nháy LED
  if (!wifiOK) {
    if (wifiConnected) {
      // Hiển thị "WiFi Lost" chỉ 1 lần
      tft.fillRect(10, 45, 220, 25, ST77XX_BLACK);
      drawDoubleBorder(10, 45, 220, 25, ST77XX_RED, ST77XX_RED, 5);
      tft.setTextSize(2); tft.setTextColor(ST77XX_RED);
      tft.setCursor(20, 120);
      tft.println("WiFi Lost! (Offline Mode)");
      wifiConnected = false;
    }

    // LED báo WiFi mất kết nối (nhấp nháy 0.5s)
    static unsigned long lastBlink = 0;
    static bool blinkState = false;
    if (millis() - lastBlink >= 500) {
      blinkState = !blinkState;
      digitalWrite(LED_WIFI_OK, LOW);
      digitalWrite(LED_WIFI_LOST, blinkState);
      lastBlink = millis();
    }

    // === 2. Tự động thử kết nối lại WiFi mỗi 10 giây ===
    static unsigned long lastRetry = 0;
    if (millis() - lastRetry >= 10000) {
      Serial.println("[WiFi] Mất kết nối, đang thử lại");
      WiFi.disconnect();
      WiFi.begin(ssid, password);  
      lastRetry = millis();
    }
  } 
  else {
    // === 3. Khi WiFi trở lại ===
    if (!wifiConnected) {
      tft.fillRect(10, 45, 220, 25, ST77XX_BLACK);
      drawDoubleBorder(10, 45, 220, 25, ST77XX_GREEN, ST77XX_GREEN, 5);
      tft.setTextSize(2); tft.setTextColor(ST77XX_GREEN);
      tft.setCursor(20, 120);
      tft.println("WiFi Connected!");
      delay(1000);
      wifiConnected = true;
    }
    digitalWrite(LED_WIFI_OK, HIGH);
    digitalWrite(LED_WIFI_LOST, LOW);
  }

  // === 4. Xử lý cảm biến & cảnh báo bình thường (kể cả khi mất WiFi) ===
  if (now - lastSensorRead >= 3000) {
    lastSensorRead = now;

    readSensors(t, h, smoke, ch4, lpg, co);
    calculateRisk(t, h, smoke, ch4, lpg, co, M_fire, M_gas);

    if (mode == 0) controlRelays(M_fire, M_gas);
    updateTFT(t, h, smoke, ch4, lpg, co, M_fire, M_gas);

    // === 5. Gửi dữ liệu Firebase nếu WiFi OK ===
    if (wifiOK && Firebase.ready() && now - lastFirebaseSend >= 5000) {
      sendToFirebase(t, h, smoke, ch4, lpg, co, M_fire, M_gas);
      lastFirebaseSend = now;
    }

    // === 6. Đồng bộ ngưỡng và chế độ mỗi 15 giây (chỉ khi có mạng) ===
    if (wifiOK && Firebase.ready() && now - lastThresholdSync >= 15000) {
      syncFireWeightsFromFirebase();   
      syncGasWeightsFromFirebase();  
      syncLevelFromFirebase();  
      syncFirebaseModeAndRelays();
      lastThresholdSync = now;
    }
  }
}
